cmake_minimum_required(VERSION 3.16)
project(charging_command_watchdog LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(charging_driver_watchdog_test tests/driver_watchdog_test.cpp)
target_compile_features(charging_driver_watchdog_test PRIVATE cxx_std_14)
target_compile_options(charging_driver_watchdog_test PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(charging_driver_watchdog_test PRIVATE include)
target_link_libraries(charging_driver_watchdog_test PRIVATE Threads::Threads util)
enable_testing()
add_test(NAME command_watchdog_pty COMMAND charging_driver_watchdog_test)
